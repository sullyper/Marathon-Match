# generate_results_html.py # # This script reads a results.csv file and generates a static HTML page # with two tables: one showing the best scores for each seed, and another # ranking players based on a point system. # # Usage: python generate_results_html.py /path/to/to/folder # If no folder is provided, it defaults to the current directory. import os import csv import sys from collections import defaultdict from itertools import groupby def read_csv_data(folder_path, names_to_skip): """ Reads results.csv and returns the data as a list of dictionaries, skipping rows for specified names. Returns None if the file is not found. """ results_file_path = os.path.join(folder_path, 'results.csv') if not os.path.exists(results_file_path): print(f"Error: '{results_file_path}' not found.") return None data = [] try: with open(results_file_path, 'r', newline='') as csvfile: reader = csv.DictReader(csvfile) for row in reader: # Skip the row if the name is in the list of names to skip if row['name'] in names_to_skip: continue # Ensure scores are treated as floats row['score'] = float(row['score']) # Convert id to int for easier handling row['id'] = int(row['id']) data.append(row) except Exception as e: print(f"Error reading {results_file_path}: {e}") return None return data def calculate_points(data): """ Calculates points for each player based on their ranks for each seed. Returns a dictionary of player points and a dictionary of player scores. """ # Point system as specified in the prompt points_list = [100, 75, 60, 50, 45, 40, 36, 32, 29, 26, 24, 22, 20, 18, 16, 15, 14, 13, 12, 11, 10, 9, 8, 7, 6, 5, 4, 3, 2] # Pad the rest with 1 point for ties points_list.extend([1] * 100) # Extend to handle more than 29 players player_scores = defaultdict(lambda: defaultdict(lambda: float('inf'))) # Get the best score for each player/id combination for row in data: name = row['name'] player_id = row['id'] score = row['score'] if score > 0 and score < player_scores[name][player_id]: player_scores[name][player_id] = score # Structure scores by id to rank them scores_by_id = defaultdict(list) for name, scores in player_scores.items(): for player_id, score in scores.items(): if player_id <= 20: # Only consider IDs 1-10 scores_by_id[player_id].append({'name': name, 'score': score}) player_points = defaultdict(lambda: {'total': 0, 'pts': defaultdict(int)}) # Calculate points for each seed (ID) for player_id in range(1, 11): if player_id in scores_by_id: # Sort players by score (lower is better) sorted_players = sorted(scores_by_id[player_id], key=lambda x: x['score']) # Group by score to handle ties and assign the same rank/points rank = 0 for score, group in groupby(sorted_players, key=lambda x: x['score']): # Assign points for this rank pts = points_list[rank] if rank < len(points_list) else 1 group_list = list(group) for player in group_list: player_points[player['name']]['pts'][player_id] = pts rank += len(group_list) # Calculate total points for each player for name, data in player_points.items(): data['total'] = sum(data['pts'].values()) return player_points, player_scores, scores_by_id def generate_html(scores_by_id, player_points, player_scores): """ Generates the HTML content for the results page. """ # Sort the players by total points in descending order for the second table sorted_players = sorted(player_points.items(), key=lambda item: item[1]['total'], reverse=True) # Get a sorted list of all unique players for the first table all_player_names = sorted(player_scores.keys()) # --- Scores by Player Table HTML (Updated) --- scores_by_name_html = f"""

Scores by Player (Lower is Better)

""" for player_id in range(1, 11): scores_by_name_html += f'' scores_by_name_html += """ """ # Find the best score for each seed to use for highlighting best_scores_per_seed = {} for player_id in range(1, 21): if player_id in scores_by_id and scores_by_id[player_id]: best_scores_per_seed[player_id] = min([p['score'] for p in scores_by_id[player_id]]) for name in all_player_names: scores_by_name_html += f""" """ for player_id in range(1, 11): score_for_id = player_scores.get(name, {}).get(player_id, 0) # Check for highlighting score_class = "" if score_for_id > 0 and player_id in best_scores_per_seed and score_for_id == best_scores_per_seed[player_id]: score_class = "text-green-600 font-bold" # Format the score to 2 decimal places formatted_score = f"{score_for_id:.2f}" if score_for_id > 0 else "N/A" scores_by_name_html += f'' scores_by_name_html += """ """ scores_by_name_html += """
NameTime {player_id}
{name} {formatted_score}
""" # --- Player Rankings Table HTML (Unchanged) --- rankings_html = f"""

Player Rankings

""" for player_id in range(1, 11): rankings_html += f'' rankings_html += """ """ current_rank = 1 # Use a set to handle tied ranks for i, (name, data) in enumerate(sorted_players): # Handle ties in rank if i > 0 and sorted_players[i][1]['total'] < sorted_players[i-1][1]['total']: current_rank = i + 1 rankings_html += f""" """ for player_id in range(1, 11): rankings_html += f'' rankings_html += "" rankings_html += """
Rank Name Total PointsPts {player_id}
{current_rank} {name} {data['total']}{data["pts"].get(player_id, 0)}
""" # --- Scores by Player Table HTML (Updated) --- extended_scores_by_name_html = f"""

Extended results (Lower is Better)

""" for player_id in range(11, 21): extended_scores_by_name_html += f'' extended_scores_by_name_html += """ """ for name in all_player_names: extended_scores_by_name_html += f""" """ for player_id in range(11, 21): score_for_id = player_scores.get(name, {}).get(player_id, 0) # Check for highlighting score_class = "" if score_for_id > 0 and player_id in best_scores_per_seed and score_for_id == best_scores_per_seed[player_id]: score_class = "text-green-600 font-bold" # Format the score to 2 decimal places formatted_score = f"{score_for_id:.2f}" if score_for_id > 0 else "N/A" extended_scores_by_name_html += f'' extended_scores_by_name_html += """ """ extended_scores_by_name_html += """
NameTime {player_id}
{name} {formatted_score}
""" full_html = f""" Racing Car Results

Racing Car Tester Results

{scores_by_name_html} {rankings_html} {extended_scores_by_name_html}
""" return full_html def main(): """ Main function to orchestrate the HTML generation process. """ if len(sys.argv) > 1: folder_path = sys.argv[1] if not os.path.isdir(folder_path): print(f"Error: The provided path '{folder_path}' is not a valid directory.") sys.exit(1) else: folder_path = os.getcwd() print("No folder path provided. Using the current directory.") # Get the optional names to skip parameter names_to_skip = [] if len(sys.argv) > 2: names_to_skip = [name.strip() for name in sys.argv[2].split(',')] print(f"Skipping the following names: {', '.join(names_to_skip)}") data = read_csv_data(folder_path, names_to_skip) if data is None: sys.exit(1) player_points, player_scores, scores_by_id = calculate_points(data) html_content = generate_html(scores_by_id, player_points, player_scores) output_file_path = os.path.join(folder_path, 'results.html') with open(output_file_path, 'w', encoding='utf-8') as f: f.write(html_content) print(f"\nSuccessfully generated HTML page at '{output_file_path}'.") if __name__ == "__main__": main()
